<script>
  var coll = document.getElementsByClassName("collapsible");
  var i;

  for (i = 0; i < coll.length; i++) {
    coll[i].addEventListener("click", function() {
      this.classList.toggle("active");
      var content = this.nextElementSibling;
      if (content.style.display === "block") {
        content.style.display = "none";
      } else {
        content.style.display = "block";
      }
    });
  }
document.querySelectorAll('.severity-table tbody tr').forEach(row => {
  const severityCell = row.cells[1];
  const value = severityCell.textContent.trim().toUpperCase();

  if (value === 'CRITICAL') {
    row.classList.add('severity-critical');
  } else if (value === 'HIGH') {
    row.classList.add('severity-high');
  } else if (value === 'MEDIUM') {
    row.classList.add('severity-medium');
  } else if (value === 'LOW') {
    row.classList.add('severity-low');
  }
});

</script>
